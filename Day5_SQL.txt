-- Day 5 — K2: SQL / RLS
begin;

-- TABELA EVENTS
create table if not exists public.events (
  id bigint generated always as identity primary key,
  title text not null,
  date text not null,
  location text not null,
  likes integer not null default 0,
  created_at timestamptz not null default now(),
  user_id uuid references auth.users(id) default auth.uid(),
  author text
);

alter table public.events
  add column if not exists user_id uuid
    references auth.users(id) default auth.uid(),
  add column if not exists author text;

alter table public.events
  alter column user_id set default auth.uid(),
  alter column likes set default 0;

-- TABELA MESSAGES
create table if not exists public.messages (
  id bigint generated always as identity primary key,
  user_id uuid not null
    references auth.users(id) default auth.uid(),
  author text not null,
  content text not null,
  created_at timestamptz not null default now()
);

alter table public.messages
  alter column user_id set default auth.uid();

-- TABELA VOTES
create table if not exists public.votes (
  id bigint generated always as identity primary key,
  user_id uuid not null
    references auth.users(id) default auth.uid(),
  option text not null
    check (option in ('Hackathon', 'Workshop', 'Turneu')),
  created_at timestamptz not null default now()
);

alter table public.votes
  alter column user_id set default auth.uid();

-- Një votë për çdo përdorues.
create unique index if not exists day5_votes_one_per_user
  on public.votes(user_id);

-- AKTIVIZIMI I RLS
alter table public.events enable row level security;
alter table public.messages enable row level security;
alter table public.votes enable row level security;

-- Heqim politikat e vjetra për këto tri tabela.
do $$
declare
  p record;
begin
  for p in
    select schemaname, tablename, policyname
    from pg_policies
    where schemaname = 'public'
      and tablename in ('events', 'messages', 'votes')
  loop
    execute format(
      'drop policy %I on %I.%I',
      p.policyname,
      p.schemaname,
      p.tablename
    );
  end loop;
end $$;

-- LEJET
revoke all privileges
  on table public.events, public.messages, public.votes
  from anon, authenticated;

grant usage on schema public to authenticated;

grant select
  on table public.events, public.messages, public.votes
  to authenticated;

grant insert
  on table public.events, public.messages, public.votes
  to authenticated;

grant update (title, date, location, likes)
  on public.events
  to authenticated;

grant delete
  on public.events
  to authenticated;

grant usage, select
  on sequence
    public.events_id_seq,
    public.messages_id_seq,
    public.votes_id_seq
  to authenticated;

-- EVENTS: LEXIMI
create policy "Day5 events read"
on public.events
for select
to authenticated
using (true);

-- EVENTS: SHTIMI NGA PËRDORUESI I KYÇUR
create policy "Day5 events insert own"
on public.events
for insert
to authenticated
with check (
  (select auth.uid()) = user_id
  and author = (select auth.jwt() ->> 'email')
  and length(btrim(title)) between 1 and 120
  and length(btrim(date)) between 1 and 80
  and length(btrim(location)) between 1 and 120
);

-- K11: NDRYSHIMI VETËM NGA PRONARI
create policy "Day5 events update own"
on public.events
for update
to authenticated
using (
  (select auth.uid()) = user_id
)
with check (
  (select auth.uid()) = user_id
  and length(btrim(title)) between 1 and 120
  and length(btrim(date)) between 1 and 80
  and length(btrim(location)) between 1 and 120
);

-- K12: FSHIRJA VETËM NGA PRONARI
create policy "Day5 events delete own"
on public.events
for delete
to authenticated
using (
  (select auth.uid()) = user_id
);

-- MESSAGES: LEXIMI
create policy "Day5 messages read"
on public.messages
for select
to authenticated
using (true);

-- MESSAGES: DËRGIMI NË EMRIN TËND
create policy "Day5 messages insert own"
on public.messages
for insert
to authenticated
with check (
  (select auth.uid()) = user_id
  and author = (select auth.jwt() ->> 'email')
  and length(btrim(content)) > 0
);

-- VOTES: LEXIMI
create policy "Day5 votes read"
on public.votes
for select
to authenticated
using (true);

-- VOTES: SHTIMI I VOTËS TËNDE
create policy "Day5 votes insert own"
on public.votes
for insert
to authenticated
with check (
  (select auth.uid()) = user_id
  and option in ('Hackathon', 'Workshop', 'Turneu')
);

commit;
